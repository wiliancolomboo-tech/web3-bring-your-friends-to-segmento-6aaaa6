FROM node:20-alpine AS builder

WORKDIR /usr/src/app
COPY package*.json ./
RUN npm ci

COPY tsconfig.json ./
COPY agent/ ./agent/
RUN npm run build:agent

FROM node:20-alpine AS runner
WORKDIR /usr/src/app

ENV NODE_ENV=production
COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

USER node
EXPOSE 8080

CMD ["node", "dist/agent/worker.js"]
